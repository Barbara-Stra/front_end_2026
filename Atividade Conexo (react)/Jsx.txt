const { useState, useEffect } = React;

const initialWords = [
  "MAÇÃ", "BANANA", "LARANJA", "UVA",
  "AZUL", "VERMELHO", "VERDE", "AMARELO",
  "CACHORRO", "GATO", "PÁSSARO", "PEIXE",
  "FUTEBOL", "BASQUETE", "VÔLEI", "TÊNIS"
];

const groups = [
  { words: ["MAÇÃ", "BANANA", "LARANJA", "UVA"], category: "Frutas", color: "#f8d568" },
  { words: ["AZUL", "VERMELHO", "VERDE", "AMARELO"], category: "Cores", color: "#6eb46e" },
  { words: ["CACHORRO", "GATO", "PÁSSARO", "PEIXE"], category: "Animais", color: "#4a9eff" },
  { words: ["FUTEBOL", "BASQUETE", "VÔLEI", "TÊNIS"], category: "Esportes", color: "#b36eff" }
];

function App() {
  const [words, setWords] = React.useState([]);
  const [selected, setSelected] = React.useState([]);
  const [foundGroups, setFoundGroups] = React.useState([]);
  const [mistakes, setMistakes] = React.useState(0);
  const [gameWon, setGameWon] = React.useState(false);

  useEffect(() => {
    setWords([...initialWords].sort(() => Math.random() - 0.5));
  }, []);

  const toggleWord = (word) => {
    if (selected.includes(word)) {
      setSelected(selected.filter(w => w !== word));
    } else if (selected.length < 4) {
      setSelected([...selected, word]);
    }
  };

  const submitGuess = () => {
    if (selected.length !== 4) return;

    const correctGroup = groups.find(group => 
      group.words.every(w => selected.includes(w)) &&
      selected.every(w => group.words.includes(w))
    );

    if (correctGroup) {
      setFoundGroups([...foundGroups, correctGroup]);
      setWords(words.filter(w => !selected.includes(w)));
      setSelected([]);

      if (foundGroups.length + 1 === 4) {
        setGameWon(true);
      }
    } else {
      setMistakes(m => m + 1);
      setSelected([]);
      if (mistakes + 1 >= 4) {
        alert("Você perdeu! Tente novamente.");
      }
    }
  };

  const shuffleWords = () => {
    setWords([...words].sort(() => Math.random() - 0.5));
  };

  return React.createElement('div', { className: 'container' },
    React.createElement('h1', null, 'Conexo'),
    React.createElement('p', null, 'Forme 4 grupos de 4 palavras que tenham algo em comum'),

    React.createElement('div', { className: 'mistakes' }, 
      `Erros: ${'❤️'.repeat(4 - mistakes)}`
    ),

    React.createElement('div', { className: 'found-groups' },
      foundGroups.map((group, i) =>
        React.createElement('div', {
          key: i,
          className: 'group',
          style: { backgroundColor: group.color }
        },
          React.createElement('strong', null, group.category),
          React.createElement('p', null, group.words.join(" • "))
        )
      )
    ),

    React.createElement('div', { className: 'grid' },
      words.map((word, index) =>
        React.createElement('button', {
          key: index,
          className: `word-btn ${selected.includes(word) ? 'selected' : ''}`,
          onClick: () => toggleWord(word)
        }, word)
      )
    ),

    React.createElement('div', { className: 'actions' },
      React.createElement('button', {
        onClick: submitGuess,
        disabled: selected.length !== 4
      }, 'Enviar'),
      React.createElement('button', { onClick: shuffleWords }, 'Embaralhar'),
      React.createElement('button', {
        onClick: () => window.location.reload()
      }, 'Novo Jogo')
    ),

    gameWon && React.createElement('h2', { className: 'win' }, '🎉 Parabéns! Você completou o Conexo!')
  );
}

ReactDOM.render(
  React.createElement(App),
  document.getElementById('root')
);
